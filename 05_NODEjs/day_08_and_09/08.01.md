NODE + MYSQL (INTRO SETUP & BASIC OPERATIONS)
=============================================================================

📌 OVERVIEW
-----------------------------------------------------------------------------
This setup connects Node.js with MySQL database to perform
basic operations like SELECT and INSERT.

Used for backend data handling in applications.


=============================================================================
SETUP (index.js)
=============================================================================

// import mysql package
const mysql = require("mysql2");

// faker → for generating dummy data
const { faker } = require('@faker-js/faker');

// uuid → for unique IDs
const { v4: uuidv4 } = require('uuid');


// create the connection to database
const connection = mysql.createConnection({
    host: 'localhost',
    user: 'root',
    database: 'demo_app',
    password: 'nithin@061'
});


=============================================================================
SELECT QUERY (READ DATA)
=============================================================================

connection.query(
    "SELECT * FROM user",
    function (err, result, fields) {
        if (err) {
            console.log(err);
            return;
        }
        console.log(result);
    }
);

// close connection (good practice)
connection.end();

=============================================================================
SELECT QUERY (WITH CONDITIONS)
=============================================================================

connection.query(
    'SELECT * FROM user WHERE name = "Page" AND age > 45',
    function (err, results, fields) {
        console.log(results);
        console.log(fields);
    }
);


=============================================================================
SELECT WITH PLACEHOLDERS (SAFE)
=============================================================================

connection.query(
    'SELECT * FROM user WHERE name = ? AND age > ?',
    ['Page', 45],
    function (err, results) {
        console.log(results);

        // close connection
        connection.end();
    }
);

👉 Placeholders (?) help prevent SQL injection.


=============================================================================
