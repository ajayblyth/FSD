8. FETCH SINGLE USER
========================

Q8. Fetch Single User by ID
-----------------------------------------------------------------------------
Given:
const users = [
  { id: 1, name: 'Sam' },
  { id: 2, name: 'Rahul' }
];

Route:
GET /users/:id

Goal:
→ return only matched user object

================================================================

======================
const express = require("express");
const app = express();

const PORT = 3000;

app.use(express.json());

app.get("/hello", (req, res) => {
  res.json({
    firstName: "ajay",
    lastName: "Sharma"
  });
});

const users = [
  { id: 1, name: "ajay" },
  { id: 2, name: "devesh" },
  { id: 3, name: "sonu" }
];

app.get("/users/:id", (req, res) => {
  const id = parseInt(req.params.id);
  console.log(id);
  const user = users.find(u => u.id === id);
  res.json({
    user
  });
});

app.get("/users", (req, res) => {
  res.json({
    users
  });
});

app.get("/user/:id", (req, res) => {
  const id = req.params.id;
  res.json({ id });
});

app.get("/user", (req, res) => {
  const name = req.query.name;
  res.json({
    name
  });
});

app.post("/users", (req, res) => {
  const name = req.body.name;
  res.json({ name });
});

app.patch("/users/:id", (req, res) => {
  const id = parseInt(req.params.id);
  const name = req.body.name;
  const user = users.find(u => u.id === id);
  if (user) {
    user.name = name;
  }
  res.json({
    user
  });
});

app.listen(PORT, () => {
  console.log(server is running on port ${PORT});
});
